public struct AedstSidebar<Header: View, Content: View, Footer: View>: View {
    public init(
        @ViewBuilder header: () -> Header,
        @ViewBuilder content: () -> Content,
        @ViewBuilder footer: () -> Footer

extension AedstSidebar where Header == EmptyView, Footer == EmptyView {
    public init(@ViewBuilder content: () -> Content)

public struct AedstSidebarGroup<Content: View>: View {

        size: AedstSidebarMenuButtonSize = .default,

}

export enum AedstSidebarVariant { default, outline }

export enum AedstSidebarSize { default, sm, lg }

export struct AedstSidebarItem {
    id: string,
    title: string,
    icon: image,
    // Short text at the reading end (a count).
    badge: string,
}

export component AedstSidebar inherits Rectangle {
    in property <[AedstSidebarItem]> items;
    in property <string> current-id;
    in property <string> title;
    in property <string> label;
    in-out property <bool> collapsed: false;
    in property <string> toggle-label;
    in property <AedstSidebarVariant> variant: AedstSidebarVariant.default;
    in property <AedstSidebarSize> size: AedstSidebarSize.default;
    callback selected(string);
    callback collapse-toggled(bool);
}

AedstSidebar {
    label: "Sections";
    title: "Workspace";
    toggle-label: "Collapse sidebar";
    variant: AedstSidebarVariant.outline;
    collapsed <=> root.rail;
    current-id: root.section;
    items: [{ id: "tasks", title: "Tasks", badge: "3" }, { id: "media", title: "Media" }];
    selected(id) => {
        root.section = id;
        GalleryState.log("sidebar:" + id);
