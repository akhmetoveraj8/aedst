import {

type DataTableColumn<TData extends RowData, TValue = unknown> = ColumnDef<
  typeof dataTableFeatures,
  TData,
  TValue
>;

interface DataTableProps<TData extends RowData> {
  /** Accessible name of the table and of its scrollable region. */
  label: string;
  /** Define at module scope or memoize: a new array rebuilds the row model. */
  columns: DataTableColumn<TData>[];
  data: TData[] | undefined;
  /** Shown in the single body row while there are no rows. */
  emptyText: string;
  isLoading?: boolean;
  /** Replaces `emptyText` while `isLoading` and there are no rows. */
  loadingText?: string;
  className?: string;
}

function DataTable<TData extends RowData>({
  label,
  columns,
  data,
  emptyText,
  isLoading = false,
  loadingText,
  className,
}: DataTableProps<TData>)

    in property <[[string]]> rows;
    in property <string> label;
    in property <string> empty-text;
    in property <string> ascending-text;
    in property <string> descending-text;
    in property <int> sort-column: -1;
    in property <bool> sort-ascending: true;
    in-out property <int> current-row: -1;
    callback sort-requested(int);
    callback row-selected(int);
    callback row-activated(int);
}
