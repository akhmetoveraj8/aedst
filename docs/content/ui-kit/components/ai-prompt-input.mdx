export enum AedstAiPromptInputStatus { ready, submitted, streaming, error }

export struct AedstAiPromptInputAttachment {
    id: string,
    title: string,
    // The size or the upload status, formatted by the caller.
    description: string,
    state: AedstAttachmentState,
    variant: AedstAttachmentVariant,
    source: image,
    // Accessible name of the chip's remove button, for example "Remove report.pdf".
    remove-label: string,
}

export component AedstAiPromptInput inherits Rectangle {
    in-out property <string> text;
    in property <string> placeholder;
    in property <string> label;
    in property <AedstAiPromptInputStatus> status: AedstAiPromptInputStatus.ready;
    in property <[AedstAiPromptInputAttachment]> attachments;
    in property <bool> enabled: true;
    in property <string> submit-label;
    in property <string> stop-label;
    in property <string> attach-label;
    in property <string> busy-label;
    out property <bool> has-focus: field.has-focus;
    out property <bool> can-submit: root.enabled && !root.busy && (root.text != "" || root.attachments.length > 0);
    callback submitted(string) -> bool;
    callback stop-requested();
    callback attach-requested();
    callback attachment-removed(string);
    public function submit();
}
