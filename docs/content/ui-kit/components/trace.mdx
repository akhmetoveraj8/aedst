```qml title="Slint"
export enum AedstTraceStatus { pending, running, success, error, skipped }

export struct AedstTraceStep {
    id: string,
    title: string,
    status: AedstTraceStatus,
    // Translated status label: the status is never shown by colour alone.
    status-text: string,
    // Already formatted by the caller, for example "1.2 s".
    duration: string,
    // Extra text (arguments, output) shown when the step is open.
    detail: string,
    // Nesting level: the caller flattens the step tree (0 for top-level steps).
    depth: int,
    open: bool,
}

export component AedstTrace inherits Rectangle {
    in property <[AedstTraceStep]> steps;
    in property <string> label;
    callback toggled(string, bool);
}
```
