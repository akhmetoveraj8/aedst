export struct AedstAiFileTreeNode {
    path: string,
    name: string,
    depth: int,
    folder: bool,
    // Folders only: shown open (chevron down, open-folder icon).
    open: bool,
    // Files only: replaces `file-icon`.
    icon: image,
    // Short text shown beside the file (never inside its button), e.g. "M" or "+12".
    badge: string,
}

export component AedstAiFileTree inherits Rectangle {
    in-out property <[AedstAiFileTreeNode]> nodes;
    in-out property <string> selected-path;
    in-out property <int> current: 0;
    in property <string> label;
    in property <image> folder-icon;
    in property <image> folder-open-icon;
    in property <image> file-icon;
    in property <string> mono-family: "monospace";
    out property <bool> has-focus: tree-focus.has-focus;
    callback toggled(string, bool);
    callback selected(string);
}
