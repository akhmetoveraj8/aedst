export enum AedstAiTestStatus { passed, failed, skipped, running }

    id: string,
    name: string,
    status: AedstAiTestStatus,
    status-label: string,
    tests: [AedstAiTest],
}

export component AedstAiTestResultsHeader inherits AedstAiChainOfThoughtHeader {
