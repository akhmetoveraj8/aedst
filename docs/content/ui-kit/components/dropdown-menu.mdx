```swift
public enum AedstMenuItemVariant { case `default`, destructive }

public enum Content { case action, checkbox, submenu, section, separator }

public struct AedstMenuItem {
    public static func action(
        _ title: String,
        systemImage: String? = nil,
        shortcut: KeyboardShortcut? = nil,
        variant: AedstMenuItemVariant = .default,
        isDisabled: Bool = false,
        action: @escaping () -> Void
    ) -> AedstMenuItem

    public static func checkbox(
        _ title: String,
        isOn: Bool,
        isDisabled: Bool = false,
        toggle: @escaping () -> Void
    ) -> AedstMenuItem

    public static func submenu(_ title: String, systemImage: String? = nil, items: [AedstMenuItem]) -> AedstMenuItem

    public static func section(_ title: String = "", items: [AedstMenuItem]) -> AedstMenuItem
}

public struct AedstMenuTriggerLabel: View {
    public init(_ title: String, systemImage: String? = nil)
}

public struct AedstDropdownMenu<Trigger: View>: View {
    public init(items: [AedstMenuItem], @ViewBuilder trigger: () -> Trigger)
}

extension AedstDropdownMenu where Trigger == AedstMenuTriggerLabel {
    public init(_ title: String, systemImage: String? = nil, items: [AedstMenuItem])
}
```

```qml title="Slint"
export enum AedstMenuItemVariant { default, destructive }

export struct AedstMenuItem {
    id: string,
    label: string,
    // Second line under the label (navigation menus).
    description: string,
    // Shown at the reading end, for example "Ctrl+K". Informational: the feature binds the key.
    shortcut: string,
    disabled: bool,
    variant: AedstMenuItemVariant,
    // Draws a separator line above this entry.
    separator: bool,
}

export component AedstMenuRow inherits Rectangle {
    in property <AedstMenuItem> data;
    in property <bool> active;
    in property <bool> checked;
    in property <bool> show-check;
    in property <int> index;
    in property <int> count;
    callback clicked();
    callback hovered();
}

export component AedstMenuList inherits Rectangle {
    in property <[AedstMenuItem]> items;
    in property <string> label;
    in property <int> checked-index: -1;
    in property <bool> autofocus: true;
    callback activated(int);
    callback dismissed();
}

export component AedstDropdownMenu inherits Rectangle {
    in property <string> text;
```
